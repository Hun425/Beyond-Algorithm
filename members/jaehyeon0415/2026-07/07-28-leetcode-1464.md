### 코드

```dart
class Solution {
  int maxProduct(List<int> nums) {
    int largest = 0;
    int secondLargest = 0;

    for (int number in nums) {
        if (number >= largest) {
            secondLargest = largest;
            largest = number;
        } else if (number > secondLargest) {
            secondLargest = number;
        }
    }

    return (largest - 1) * (secondLargest - 1);
  }
}
```
